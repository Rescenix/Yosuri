### 1. `web_fetch` — 网页内容抓取 MCP 服务器

**新建文件**：`main-backend/mcp/web_fetch_server.py`

**实现**：
```python
from mcp.server import Server
import httpx
from html.parser import HTMLParser

class TextExtractor(HTMLParser):
    def __init__(self):
        super().__init__()
        self.text = []
        self.skip = False
    def handle_starttag(self, tag, attrs):
        if tag in ('script', 'style', 'nav', 'footer', 'header'):
            self.skip = True
    def handle_endtag(self, tag):
        if tag in ('script', 'style', 'nav', 'footer', 'header'):
            self.skip = False
    def handle_data(self, data):
        if not self.skip:
            self.text.append(data.strip())

@server.tool()
async def web_fetch(url: str, max_chars: int = 8000) -> str:
    """抓取 URL 网页内容，返回纯文本。"""
    async with httpx.AsyncClient(timeout=30) as client:
        resp = await client.get(url, headers={"User-Agent": "Mozilla/5.0"})
        resp.raise_for_status()
    extractor = TextExtractor()
    extractor.feed(resp.text)
    text = "\n".join(t for t in extractor.text if t)
    if len(text) > max_chars:
        text = text[:max_chars] + "\n...(已截断)"
    return text
```

### 2. `screenshot` — 网页截图 MCP 服务器

**新建文件**：`main-backend/mcp/screenshot_server.py`

**实现**：Playwright 截图 → 视觉模型分析（用 GLM-5.2 或 Qwen-VL-Max）

```python
from mcp.server import Server
from playwright.async_api import async_playwright
import base64, os, httpx

# 视觉模型配置（优先用 GLM-5.2，fallback 到 Qwen-VL-Max）
SF_API_KEY = os.getenv("SILICONFLOW_API_KEY", "")
SF_MODEL = "zai-org/GLM-5.2"  # SiliconFlow 上的视觉模型

@server.tool()
async def screenshot(url: str) -> str:
    """截取网页截图并分析内容。"""
    async with async_playwright() as p:
        browser = await p.chromium.launch()
        page = await browser.new_page()
        await page.goto(url, wait_until="networkidle")
        screenshot_bytes = await page.screenshot(full_page=False)
        await browser.close()
    
    img_b64 = base64.b64encode(screenshot_bytes).decode()
    
    # 用视觉模型直接分析截图
    async with httpx.AsyncClient(timeout=60) as client:
        resp = await client.post(
            "https://api.siliconflow.cn/v1/chat/completions",
            headers={"Authorization": f"Bearer {SF_API_KEY}"},
            json={
                "model": SF_MODEL,
                "messages": [{
                    "role": "user",
                    "content": [
                        {"type": "text", "text": "请详细分析这张网页截图：页面内容、布局结构、关键元素、可交互组件。"},
                        {"type": "image_url", "image_url": {"url": f"data:image/png;base64,{img_b64}"}}
                    ]
                }],
                "max_tokens": 2000
            }
        )
        data = resp.json()
        return data["choices"][0]["message"]["content"]
```

### 3. `view_image` — 图片分析 MCP 服务器（支持多轮追问）

**新建文件**：`main-backend/mcp/view_image_server.py`

**关键改进**：不用 Gemini 预处理，用视觉模型直接看图。支持传入之前的问题历史，实现多轮追问。

```python
from mcp.server import Server
import base64, os, httpx, json

SF_API_KEY = os.getenv("SILICONFLOW_API_KEY", "")
SF_MODEL = "zai-org/GLM-5.2"

@server.tool()
async def view_image(
    image_url: str = "",
    image_base64: str = "",
    question: str = "请详细描述这张图片的内容",
    history: str = ""  # JSON 格式的之前问答历史，支持多轮追问
) -> str:
    """
    查看并分析图片。支持多轮追问。
    
    用法示例：
    1. 首次分析：view_image(image_url="...", question="这个UI是什么应用？")
    2. 追问细节：view_image(image_url="...", question="左边的菜单有哪些选项？", 
                           history='[{"q":"这个UI是什么应用？","a":"这是一个代码编辑器..."}]')
    3. 深入分析：view_image(image_url="...", question="代码区域的语法高亮有什么问题？",
                           history='[{"q":"这个UI是什么应用？","a":"代码编辑器"}, {"q":"左边菜单？","a":"文件、编辑、视图..."}]')
    """
    if image_url and not image_base64:
        async with httpx.AsyncClient(timeout=30) as client:
            resp = await client.get(image_url)
            image_base64 = base64.b64encode(resp.content).decode()
    
    # 构建多轮对话消息
    messages = []
    
    # 加入历史问答（如果有）
    if history:
        try:
            hist_list = json.loads(history)
            for h in hist_list:
                messages.append({"role": "user", "content": h["q"]})
                messages.append({"role": "assistant", "content": h["a"]})
        except:
            pass
    
    # 当前问题 + 图片
    messages.append({
        "role": "user",
        "content": [
            {"type": "text", "text": question},
            {"type": "image_url", "image_url": {"url": f"data:image/jpeg;base64,{image_base64}"}}
        ]
    })
    
    async with httpx.AsyncClient(timeout=60) as client:
        resp = await client.post(
            "https://api.siliconflow.cn/v1/chat/completions",
            headers={"Authorization": f"Bearer {SF_API_KEY}"},
            json={
                "model": SF_MODEL,
                "messages": messages,
                "max_tokens": 3000
            }
        )
        data = resp.json()
        return data["choices"][0]["message"]["content"]
```

### 4. `generate_image` — 图片生成 MCP 服务器

**新建文件**：`main-backend/mcp/generate_image_server.py`

```python
from mcp.server import Server
import os, httpx, asyncio

DASHSCOPE_API_KEY = os.getenv("DASHSCOPE_API_KEY", "")

@server.tool()
async def generate_image(prompt: str) -> str:
    """根据文字描述生成图片。"""
    async with httpx.AsyncClient(timeout=60) as client:
        resp = await client.post(
            "https://dashscope.aliyuncs.com/api/v1/services/aigc/text2image/image-synthesis",
            headers={
                "Authorization": f"Bearer {DASHSCOPE_API_KEY}",
                "X-DashScope-Async": "enable"
            },
            json={
                "model": "qwen-image-plus",
                "input": {"prompt": prompt},
                "parameters": {"size": "1024*1024", "n": 1}
            }
        )
        task_id = resp.json()["output"]["task_id"]
        
        for _ in range(30):
            await asyncio.sleep(3)
            resp = await client.get(
                f"https://dashscope.aliyuncs.com/api/v1/tasks/{task_id}",
                headers={"Authorization": f"Bearer {DASHSCOPE_API_KEY}"}
            )
            data = resp.json()
            if data["output"]["task_status"] == "SUCCEEDED":
                return f"图片已生成: {data['output']['results'][0]['url']}"
            elif data["output"]["task_status"] == "FAILED":
                return f"生成失败: {data['output'].get('message', '未知错误')}"
        
        return "生成超时（60秒）"
```

### 5. Subagent 工具扩展

**修改文件**：`main-backend/internal/handler/subagent.go`

```go
// 扩展子代理工具白名单
func isSubagentToolAllowed(name string) bool {
    if subagentAllowedTools[name] { return true }
    if strings.HasPrefix(name, "mcp__") {
        // 允许搜索、抓取、视觉类 MCP 工具
        allowed := []string{"grep", "web_fetch", "view_image", "screenshot"}
        for _, a := range allowed {
            if strings.Contains(name, a) { return true }
        }
    }
    return false
}
```

## 视觉模型选择

| 模型 | 端点 | 优势 |
|------|------|------|
| **GLM-5.2** (SiliconFlow) | `api.siliconflow.cn/v1` | 已有 key，免费，支持视觉 |
| **Qwen-VL-Max** (DashScope) | `dashscope.aliyuncs.com` | 已有 key，中文理解强 |
| **Gemini 2.5 Flash** (Google) | `generativelanguage.googleapis.com` | fallback，网络可能不通 |

优先用 GLM-5.2（SiliconFlow），网络稳定且免费。
